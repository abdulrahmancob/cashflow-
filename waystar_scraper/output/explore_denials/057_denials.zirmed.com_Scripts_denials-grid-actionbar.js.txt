// CLAIM EDITOR POPUP //

function OpenClaimEditor(claimID, claimType, isDraft)
{
    if (typeof claimsInterval === 'undefined') {
        var scrollbars;
        if (screen.availWidth <= 1024)
        {
            scrollbars = 'scrollbars=1,';
        } else
        {
            scrollbars = '';
        }

        if (claimType == "Professional")
        {
            winEdit = open(ClaimsSiteURL + 'Editor/V5010/Professional/Main.aspx?sec=0&draft=' + isDraft + '&editclaimid=' + claimID, 'Edit', 'height=700,width=1010,' + scrollbars + 'resizable=1,status=0,location=0, menubar=0,top=0,left=0');
        }
        else
        {
            winEdit = open(ClaimsSiteURL + 'Editor/V5010/Institutional/Main.aspx?sec=0&draft=' + isDraft + '&editclaimid=' + claimID, 'Edit', 'height=700,width=1010,' + scrollbars + ' resizable=1,status=0,location=0, menubar=0,top=0,left=0');
        }

        // polls to check if the resubmit window has been closed
        claimsInterval = setInterval(hasUserClosedEditor, 100);
    }
}

var attemptsCheckedForClaimQueue = 0;

function hasUserClosedEditor() {                        

    if (winEdit.closed) {                        

        clearInterval(claimsInterval);        
        claimsInterval = undefined;
        attemptsCheckedForClaimQueue = 0;

        // very likely it won't be done right away.  pause before hitting server
        setTimeout(isClaimQueuedForResubmission, 100);        
    }
}

function isClaimQueuedForResubmission() {                    
    
    var data = {
        'denialId': denialQueuedId
    };        

    $.ajax({
        type: "POST",
        url: ClaimQueuedForResubmitURL,
        data: data,        
        cache: false,
        traditional: false,
        success: function (data) { 

            if (data.queued) {

                updateStatusToQueued(denialQueuedId);
                                
                attemptsCheckedForClaimQueue = 0;

            } else if (attemptsCheckedForClaimQueue < 10) {

                setTimeout(isClaimQueuedForResubmission, 1500);

                attemptsCheckedForClaimQueue++;
            }
        }
    });       
}

function updateStatusToQueued(denialId) {

    // the stage column can be in different places in workcenter, so this finds the index of the column first
    var headerColumns = $(".gridViewHeader th");

    var stageIndex = headerColumns.index($(".gridViewHeader th:contains('Stage')").not(':contains("Days in")'));

    var stageElement = $(".gridViewExpandableRow[data-denialid=" + denialId + "] td")[stageIndex];    

    $(stageElement).html("Queue'd for Resubmission");

    return false;
}

function routeToAppeals(denialID, payerID) {
    var url = AppealURL + "?id=" + denialID;
    
    if (IsWorkcenter) {
        url += "&appId=41&tabAppId=40&returntoworkcenter=true";
    }
    else if (IsDenialSearch)
    	url += "?returntodenialsearch=true";

    if (payerID != null) {
        url += "&payerID=" + payerID;
    }

    var isElectronicSubmission = $(rowLastHovered).attr('data-iselectronicsubmission');
    if (isElectronicSubmission && isElectronicSubmission.toLowerCase() === 'true') {
        url += "&electronicSubmission=true";
    }

    window.location.href = url;
    return false;
}

// ELIGIBILITY INQUIRY //

function onEligibilityBtnClick()
{
    $("#loadingouter").show();

    $(denialEligibility).attr("disabled", true);

    var denialID = $(rowLastHovered).data("denialid");
    var eligibilityURL = "";
    var tokenID = 0;

    var data =
    {
        'denialID': denialID
    };

    $.ajax({
        type: "POST",
        url: EligibilityInquiryURL,
        data: data,
        cache: false,
        traditional: false,
        success: function (data)
        {
            tokenID = data.tokenID;
            eligibilityURL = data.url;

            if (tokenID > 0 && eligibilityURL != null && eligibilityURL != "")
            {
                var denialRow = $(".tagRow[data-denialid=" + denialID + "]");
                $(denialRow).attr("data-tokenid", tokenID);

                if (CurrentEnvironment != null && parseInt(CurrentEnvironment) > 1)
                {
                    //document.domain = "zirmed.com";
                    document.domain = getDocumentDomain(true);
                }

                /// 368099 Adding event listener since latest Chrome disallows modifying document.domain with scripts (on Elig side).
                /// https://developer.chrome.com/blog/document-domain-setter-deprecation/

                var FromDenials = window.open(eligibilityURL, "FromDenials", "width=1050,height=750,scrollbars=1,resizable=1,menubar=0,toolbar=1,status=1");
                $(denialEligibility).removeAttr("disabled");

                FromDenials.addEventListener('message', (event) => {

                    // Reject all messages except ones from eligibility.
                    if (!event.origin.includes("eligibility.zirmed.com")) return;

                    // Filter success messages. Eligibility will return the TokenID which is an integer.
                    if (Number.isInteger(event.data)) {
                        elgCallback(event.data);
                    }
                });

            }

            setTimeout(function () {
                $("#loadingouter").hide();
            }, 250);

            // use timeout below to test callback function...

            //setTimeout(function () {
            //    elgCallback(tokenID);
            //}, 5000);
        }
    });

    return false;
}

function elgCallback(tokenID)
{
    var elgPending = $(".tagRow[data-tokenid=" + tokenID + "] .elg-pending");
    elgPending.removeClass("hide-content");
}

function getDocumentDomain(excludeDot) {
    var host = document.domain;
    var lastDot = host.lastIndexOf('.');
    if (lastDot > -1) {
        var secondToLastDot = host.substring(0, lastDot).lastIndexOf('.');
        if (secondToLastDot > -1) {
            return excludeDot ? host.substring(secondToLastDot + 1) : host.substring(secondToLastDot);
        }
    }
    return host;
}
