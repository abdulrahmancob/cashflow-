function removeDenialUnderUserAction() {
    var denialId = denialIDReview;

    var data = {
        'denialID': denialId,
    };

    $.ajax({
        url: RemoveDenialUnderActionURL,
        data: data,
        cache: false,
        traditional: false,
        type: "POST",
        success: function (data) {
            $("#loadingouter").hide();
        }
    });

    // must  have a return to ensure execution
    // anything other than undefined will create a prompt in IE
    return undefined;
};

function ConfigureOverviewOnClicks() {

        var disabled = $(closedRows).css('display') == 'none';
        if (disabled) {

            $(clickedLink).html("Collapse");
        } else {

            $(clickedLink).html("Show All " + lines + " Lines");
        }
        return false;

        var hideTableLink = '.inactiveTableHide' + id;

            $(showTableLink).hide();

        var clickedLink = $(this);

        var enteredDate = $("#editReminderText").val();

        if (ValidateReviewReminderDate(false)) {
            var denialID = denialIDReview;
            var data = {
                'date': enteredDate,
                'denialID': denialID
            };

            $.ajax({
                type: "POST",
                url: ReminderDateURL,
                data: data,
                cache: false,
                traditional: false,
                success: function (data) {
                    if (data) {
                        $("#editReminder").hide();

                        if (enteredDate.toLocaleLowerCase() === "mm/dd/yyyy") {
                            $("#displayReminderNoDate").show();
                            addHint($("#editReminderText"));
                            $("#currentReminder").val("mm/dd/yyyy");
                        }
                        else {
                            $("#displayReminder").show();
                            $("#reminderDateData").html(enteredDate);
                            $("#editReminderText").val(enteredDate);
                            $("#currentReminder").val(enteredDate);
                        }
                    }
                    else {
                        ReminderDateError(true);
                    }
                }
            });
        }
    });

    $("#reminderDateCancel").on("click", function () {
        var date = $("#currentReminder").val();

        $("#editReminder").hide();
        if (date === "mm/dd/yyyy") {
            $("#displayReminderNoDate").show();
            addHint($("#editReminderText"));
        } else {
            $("#displayReminder").show();

    $("#editRootCauseBtn").on("click", function () {
        var rootCauseText = $("#rootCauseSelect").children(':selected').text();
        if (rootCauseText != "Select" && $("#rootCauseSelect").val() != 0) {
            var data = {
                denialId: denialIDReview,
                custId: $(this).attr('data-custid'),
                rootCauseID: $("#rootCauseSelect").val(),
                rootCauseNoteID: $("#hdnNotesId").val(),
                rootCauseNoteText: $("#rootCauseNoteText").val()
            }

            $.ajax({
                url: SaveRootCauseURL,
                data: data,
                type: "POST",
                cache: false,
                traditional: false,
                success: function (returnData) {
                    var selected = $("#rootCauseSelect").find("option:selected");
                    var name = returnData.name;
                    var description = returnData.desc;
                    var text = $("#rootCauseNoteText").val();
                    $("#editRootCause").hide();
                    $(".rootCauseNoteText").hide();
                    $("#rootCauseName").html(name);
                    $(".overviewRootCauseLink").removeClass("displayRootCause");
                    $("#upperFooterNote").html("Note from " + returnData.user + ":");
                    $("#rootCauseAssigned").css("display", "inline-block");
                    if (text != "") {
                        $("#lowerFooterNote").html(text);
                        $(".rootCauseNoteText").show();
                        $(".rootCauseNoteContainer").show();
                        $(".rootCauseNote").hide();
                    }
                    else {
                        $("#lowerFooterNote").html("");
                        $(".rootCauseNoteText").hide();
                        $(".rootCauseNoteContainer").hide();
                        $(".rootCauseNote").show();
                    }
                    if (description != null && description != "") {
                        $("#rootCauseDescription").html(description);
                    }
                    if (returnData.noteId !== null) {
                        $("#hdnNotesId").val(returnData.noteId);
                    }
                    $("#rootcauseIDHdn").val(selected.val());
                    $("#rootcauseNoteHdn").val(text);
                    $("#rootCauseAssigned").show();
                    $(".displayRootCause").show();
                },
                error: function (jqXHR, textStatus, errorThrown) {
                    handleAjaxError(jqXHR, textStatus, errorThrown);
                }
            });
        } 

        return false;
    });

        } else {
            var userSettings = {
                'saveValue': false,

        }

        ReorderList($('#activitesToday > div'));

        ReorderList($('#activitiesOlder > div'));

function subtractDays(dtToday, numDays) {
    return new Date(
        dtToday.getFullYear(),
        dtToday.getMonth(),
        dtToday.getDate() - numDays,
        dtToday.getHours(),
        dtToday.getMinutes(),
        dtToday.getSeconds(),
        dtToday.getMilliseconds()
    );
}
function getLastXXDays(numDays) {
    var dtToday = new Date();
    var dtSubtracted = subtractDays(dtToday, numDays);
    var newStart = zeroPad(dtSubtracted.getMonth() + 1, 2) + "/" + zeroPad(dtSubtracted.getDate(), 2) + "/" + dtSubtracted.getFullYear();
    var newEnd = zeroPad(dtToday.getMonth() + 1, 2) + "/" + zeroPad(dtToday.getDate(), 2) + "/" + dtToday.getFullYear();
    toggleDatePickers(newStart, newEnd);
}

function validateForm() {
    var reportingPeriod = $("#reportingperiodInput option:selected").val();
    var startDate = $("#StartDate").val();
    var endDate = $("#EndDate").val();
    var errorMsg = "";
    var isOK = true;
    var MaxRange = 1460;

    if (reportingPeriod == "") {
        isOK = false;
        errorMsg += "You must select a Reporting Period<br>";
    }
    if (startDate == "" || startDate == "mm/dd/yyyy") {
        isOK = false;
        errorMsg += "You must select a Start Date<br>";
    }
    if (endDate == "" || endDate == "mm/dd/yyyy") {
        isOK = false;
        errorMsg += "You must select a End Date<br>";
    }

    var dateOffset = findDateOffset(startDate, endDate);
    if (MaxRange < dateOffset) {
        isOK = false;
        errorMsg += "The Reporting Period date range is limited to " + MaxRange + " days.<br>";
    }

    if (errorMsg.length > 0) {
        DisplayErrorModal(errorMsg, "Error");
    }

    return isOK;
}

function findDateOffset(firstDate, secondDate) {
    var mdy = firstDate.split('/');
    var firstDateComp = new Date(mdy[2], mdy[0] - 1, mdy[1]);
    var mdy = secondDate.split('/');
    var secondDateComp = new Date(mdy[2], mdy[0] - 1, mdy[1]);
    return Math.round((secondDateComp - firstDateComp) / (1000 * 60 * 60 * 24));
}
function zeroPad(num, places) {
    var zero = places - num.toString().length + 1;
    return Array(+(zero > 0 && zero)).join("0") + num;
}

    $.ajax({
        url: PollDenialTagsURL,
        data: data,
        cache: false,
        traditional: false,
        type: "POST",
        success: function (data) {
            if (data.taggerRan) {
                $("#overviewDenialTags").html(data.tagsPartial);
                clearInterval(pollTimer);
                pollTags = 'False';
                $("#btnContinueMerge").prop('disabled', false);       
            }
            else {
                if (pollCount == 15) {
                    clearInterval(pollTimer);
                }
                pollCount++;
            }
        }
    });
}

function ReorderList(list) {
    for (var i = 0, length = list.length; i < length; i++) {
        $(list[i]).prependTo(list[i].parentNode);
    }
}

function setupActivityOrdering() {
    var oldToNew = $('.activityDropDown').val() === 'old';
    if (oldToNew) {
        ReorderList($('#activityList > div'));
    }
}

function getDocumentDomain(excludeDot) {
    var host = document.domain;
    var lastDot = host.lastIndexOf('.');
    if (lastDot > -1) {
        var secondToLastDot = host.substring(0, lastDot).lastIndexOf('.');
        if (secondToLastDot > -1) {
            return excludeDot ? host.substring(secondToLastDot + 1) : host.substring(secondToLastDot);
        }
    }
    return host;
}
