function toggleCheckboxes(changed, other) {
    if ($(changed).is(':checked')) {
        $(other).prop('disabled', true);
    } else {
        $(other).prop('disabled', false);
    }
}

function setUpAdvancedSearchModal() {
    setupControls($("#advancedSearchModal"));

    $("#filterCategories ul li a").on('click', function () {
        selectFilterType($(this));
        return false;
    });

    var $multiSelectListItems = $("#filterOptions .multiSelect li label");

    $multiSelectListItems.children("input[type='checkbox'],input[type='radio']").on("click", function (e) {
        advancedMultiSelectDropDownClick($(this));
    });

    $("#optionContainer .textInput").on("change", function () {
        var filterGroup = null;
        var isFrom = null;
        var itemID = $(this).attr("id");
        if (itemID == "txtPatientNameModal" || itemID == "txtClaimNumberModal" || itemID == "txtDenialIDModal") {
            filterGroup = $(this).closest("ul").attr("id").split("_")[1];
        }
        if (itemID.indexOf("Date") > -1) {
            filterGroup = $(this).closest("ul").attr("id").split("_")[1];
            isFrom = itemID.indexOf("From") > -1;
        }
        if (itemID.indexOf("Amount") > -1) {
            filterGroup = $(this).closest("ul.multiSelect").attr("id").split("_")[1];
            isFrom = itemID.indexOf("Min") > -1;
        }
        removeTagItItem(filterGroup, null, isFrom);

        var selection = $("#" + itemID).val();

        if (selection != "") {
            if (itemID.indexOf("Amount") > -1) {
                selection = "$" + selection;
            }
            if (itemID.indexOf("From") > -1 || itemID.indexOf("Min") > -1) {
                if (itemID === "WorkcenterArgs_DenialDateFrom") {
                    var denialDateFrom = $("#WorkcenterArgs_DenialDateFrom").datepicker('getDate');
                    if (denialDateFrom < MinAllowedDateForDenialSearch) {
                        $('#WorkcenterArgs_DenialDateFrom').val(MinAllowedDateForDenialSearch.toLocaleDateString());
                        selection = MinAllowedDateForDenialSearch.toLocaleDateString();
                    }
                }
                selection = "From " + selection;
            }
            else if (itemID.indexOf("To") > -1 || itemID.indexOf("Max") > -1) {
                selection = "To " + selection;
                if (itemID === "WorkcenterArgs_DenialDateTo") {
                    var denialDateTo = $('#WorkcenterArgs_DenialDateTo').datepicker('getDate');
                    if (denialDateTo < MinAllowedDateForDenialSearch) {
                        $('#WorkcenterArgs_DenialDateTo').val(MinAllowedDateForDenialSearch.toLocaleDateString());
                        selection = MinAllowedDateForDenialSearch.toLocaleDateString();
                    }
                }
            }
            createTagItItem(selection, selection, filterGroup, "");
        }
    });

    var deniedAmountMin = $("#WorkcenterArgs_DeniedAmountMin").val();
    var deniedAmountMax = $("#WorkcenterArgs_DeniedAmountMax").val();

    $("#Options_DeniedAmount > li > input").each(function () {
        if ($(this).hasClass("all") && deniedAmountMin == "" && deniedAmountMax == "") {
            $(this).parent().trigger("click");
            return false;
        }
        else if (deniedAmountMin != "" || deniedAmountMax != "") {
            if ($(this).hasClass("customDeniedAmount")) {
                $(this).parent().trigger("click");
                $("#customDeniedAmount").show();
                return false;
            }
            else {
                if ($(this).data("min") != undefined) {
                    if ($(this).data("max") != undefined) {
                        if ($(this).data("min") == deniedAmountMin && $(this).data("max") == deniedAmountMax) {
                            $(this).parent().trigger("click");
                            return false;
                        }
                    }
                    else {
                        if ($(this).data("min") == deniedAmountMin && deniedAmountMax == "") {
                            $(this).parent().trigger("click");
                            return false;
                        }
                    }
                }
            }
        }
    });

    $("#Options_DeniedAmount li input[type='radio']").on('change', function () {
        var $deniedAmountMin = $("#WorkcenterArgs_DeniedAmountMin");
        var $deniedAmountMax = $("#WorkcenterArgs_DeniedAmountMax");

            $("#customDeniedAmount").show();

        }

            }
            if ($(this).data("max") != undefined) {

function BulkReassign($closeButton) {
    $closeButton.button("disable");

    var offset = $closeButton.offset();

    var panel = $('#bulkCloseLoadingPanel');

    panel.css({ top: offset.top + 3, left: offset.left - 30 });

    panel.show();

    var reassignNote = "";
    var workgroupName = $('#WorkcenterArgs_WorkgroupID option:selected').text();

    if ($.trim($('#bulkReassignNoteTextArea').val()) === '') {
        reassignNote = "Denial moved from " + workgroupName + " to " + $("#ddl_bulkReassignWorkGroup option:selected").text();
    } else {
        reassignNote = $("#bulkReassignNoteTextArea").val();
    }

    $.ajax({
        url: ReassignURL,
        data: {
            oldWorkgroupId: $('#WorkcenterArgs_WorkgroupID option:selected').val(),
            workgroupName: workgroupName,
            newWorkgroupId: $("#ddl_bulkReassignWorkGroup").val(),
            note: reassignNote
        },
        traditional: true,
        cache: false,
        type: "POST",
        success: function () {
            location.reload();
            $(this).dialog('close');
            toggleLoading(false);
        }
    });
}

function BulkClose($closeButton) {
    $closeButton.button("disable");

    var offset = $closeButton.offset();

    var panel = $('#bulkCloseLoadingPanel');

    panel.css({ top: offset.top + 3, left: offset.left - 30 });

    panel.show();

    $.ajax({
        url: CloseURL,
        data: {
            workgroupId: $('#WorkcenterArgs_WorkgroupID option:selected').val(),
            closeType: $("#closeTypesDD_").val(),
            closeNote: $('#closeNoteTextArea').val()
        },
        traditional: true,
        cache: false,
        type: "POST",
        success: function (result) {
            if (result.success == true) {
                PollForPendingClosings(result.requestID);
            }
            else {
                alert('An error occurred while attempting to close denials');

                $closeButton.button('enable');

                panel.hide();
            }
        },
        error: function () {
            alert('An error occurred while attempting to close denials.');
        }
    });
}

function PollForPendingClosings(requestID) {
    var maxAttempts = 15;
    var currentAttempt = 0;
    var timer = setTimeout(function poll() {
        $.ajax({
            url: PollForPendingClosingsURL,
            data: { requestID: requestID },
            type: "POST",
            success: function (result) {
                if (!result.denialsQueued && !result.requestErrored && currentAttempt != maxAttempts) {
                    currentAttempt++;
                    setTimeout(poll, 1000);
                }
                else if (result.requestErrored) {
                    alert('An error occurred while attempting to close denials.');
                }
                else {
                    $('#bulkCloseLoadingPanel').hide();
                    if ($("#bulkCloseModal").is(':visible')) {
                        $("#bulkCloseModal").dialog('close');
                    }
                    if ($("#areYouSureModal").is(':visible')) {
                        $("#areYouSureModal").dialog('close');
                    }
                    reloadWorkcenterGrid();
                }
            }
        });
    }, 1000);
}

function CheckMultiplePayers(denialIDs) {
    $.ajax({
        url: CheckMultiplePayersURL,
        data: JSON.stringify(denialIDs),
        type: "POST",
        traditional: true,
        contentType: "application/json; charset=utf-8",
        success: function (data) {
            if (data == "") {
                if (denialIDs.length > parseInt($('#AppealBatchLimit').val())) {
                    $("#appealLimitModal").dialog({
                        modal: true,
                        resizable: false,
                        width: 500,
                        title: "Too Many Denials Selected",
                        buttons: {
                            "Okay": {
                                click: function () { $(this).dialog('close'); },
                                text: "Okay"
                            }
                        }
                    });
                }
                else {
                    checkExistingDrafts(denialIDs.join(), null);
                }
            }
            else {
                $("#appealPayersModal").html(data);
                openBulkAppealPayerSelectModal();
            }
        }
    });
}

function checkExistingDrafts(denialIDs, payerID) {
    var data = {
        denialIDs: denialIDs,
        payerID: payerID
    };
    $.ajax({
        url: CheckExistingDraftsURL,
        data: JSON.stringify(data),
        type: "POST",
        traditional: true,
        contentType: "application/json; charset=utf-8",
        success: function (data) {
            if (data == "") {
                bulkRouteToAppeals(denialIDs, payerID);
            }
            else {
                $("#existingDraftsModal").html(data);
                $("#existingDraftsModal").dialog({
                    title: "Existing Appeals",
                    modal: true,
                    resizable: false,
                    width: 450,
                    close: function () {
                        $("#existingDraftsModal").empty();
                    },
                    buttons: {
                        "Continue Appeals": {
                            click: function () {
                                bulkRouteToAppeals(denialIDs, payerID);
                            },
                            text: "Continue Appeals"
                        },
                        "Remove from Batch": {
                            click: function () {
                                var denialsWithoutExistingDrafts = $("#denialsWithoutExistingDrafts").val();
                                if (denialsWithoutExistingDrafts.length > 0) {
                                    bulkRouteToAppeals(denialsWithoutExistingDrafts, payerID);
                                }
                                else {
                                    $(this).dialog('close');
                                }

                            },
                            text: "Remove from Batch"
                        }
                    }
                });
            }
        }
    });
}

function bulkRouteToAppeals(denialIDs, payerID) {
    var url = CheckAppealStepURL + "?denialIDs=" + denialIDs;

    if (IsWorkcenter) {
        url += "&appId=41&tabAppId=40&returntoworkcenter=true";
    }

    if (payerID != null) {
        url += "&payerID=" + payerID;
    }

    window.location.href = url;
    return false;
}

function NeedSelection(func) {
    var selectedWorkgroupId = $('#WorkcenterArgs_WorkgroupID option:selected').val();

    if (selectedWorkgroupId == null) {
        showAlert('No workgroup is selected. Please select a workgroup before continuing.', '');
        return false;
    }
    else {
        $.ajax({
            url: ValidateDenialsToWorkURL,
            data: {
                workgroupId: selectedWorkgroupId
            },
            traditional: true,
            cache: false,
            type: "POST",
            success: function (data) {
                if (data == null || data == 0) {
                    showAlert('No denials have been selected. Please select denials before continuing.', '');
                    return false;
                }
                else {
                    selectedCount = data;
                    func();
                }
            }
        });
    }
}

        workflowStages.push($(this).val());
    });
    var categories = new Array();
    $("#categoryDropDown").children("ul").children("li:not(.selectAll)").children("input[type='checkbox']:checked").each(function () {
        categories.push($(this).val());
    });

            $checkededPayers = $checkededPayers.not("[data-filterselection='" + $payerIDArg.val() + "']");

        });

            $txtPayer.val("");
        }

        }

    var billingProviders = getFiltersForSearch("BillingProvider");
    var renderingProviders = getFiltersForSearch("RenderingProvider");
    var carcCodes = getFiltersForSearch("CarcCodes");
    var rarcCodes = getFiltersForSearch("RarcCodes");
    var denialId = getFiltersForSearch("DenialID");
    if (denialId.length > 0) {
        denialId = denialId[0];
    }
    else {
        denialId = "";
    }

    var data = {
        args: {
            WorkgroupID: $("#WorkcenterArgs_WorkgroupID").val(),
            WorkgroupName: $('#WorkcenterArgs_WorkgroupID option:selected').text(),
            WorkcenterStages: workflowStages,
            CategoryIDs: categories,
            PatientName: $("#txtPatientName").val().trim(),
            ClaimNumber: $("#txtClaimNumber").val().trim(),
            DenialID: denialId,
            PayerIDs: payerIDs,
            Payer: $txtPayer.val().trim(),

            Account: $txtAccount.val().trim(),

function CaptureSelectedDenial($checkbox) {
    var denialId = $checkbox.closest("tr").data("denialid");
    var isChecked = SelectingAll ? !$checkbox.is(':checked') : $checkbox.is(':checked');

    if ($checkbox.is(':checked')) {
        $checkbox.closest("tr").addClass('selectedRow');
    }
    else {
        $checkbox.closest("tr").removeClass('selectedRow');
    }

    var data = {
        denialId: denialId,
        selected: isChecked
    }

    $.ajax({
        url: CaptureSelectedDenialURL,
        data: data,
        traditional: true,
        cache: false,
        type: "POST",
        success: function (result) {
            selectedCount = result;
        }
    });
}

function CaptureDenialsSelection($checkbox, $option) {
    var selectionType = $option.data("selectiontype");

    switch (selectionType) {
        case 1:
            PurgeDenialSelection(true);
            break;
        case 2:
            PurgeDenialSelection(false);
            break;
        case 3:
            PageSelection(true);
            break;
        case 4:
            PageSelection(false);
            break;
    }

    if (SelectingAll) {
        $checkbox.prop('checked', true);
    }
    else if ($option.data("select") == true) {
        $checkbox.prop('checked', true);
    }
    else {
        $checkbox.prop('checked', false);
    }
}

function PurgeDenialSelection(selectingAll) {

        data: data,
        traditional: true,
        cache: false,
        type: "POST",

        }
    });
}

function setPagerStyles() {
    var currentPage = parseInt($('#CurrentPageInput').val());
    var totalPages = parseInt($('#lblTotalpages').text());
    $('#ctrl_ItemsPerPage').val($("#pageSize").val());

    if (currentPage <= 1) {
        $('.pagerFirstImage').addClass('disabled');
        $('.pagerPreviousImage').addClass('disabled');
        $('#Pager_Top_Prev').addClass('disabled');
        $('#Pager_Btm_First').addClass('disabled');
        $('#Pager_Btm_Prev').addClass('disabled');
    }
    if (currentPage > 1) {
        $('.pagerFirstImage').removeClass('disabled');
        $('.pagerPreviousImage').removeClass('disabled');
        $('#Pager_Top_Prev').removeClass('disabled');
        $('#Pager_Btm_First').removeClass('disabled');
        $('#Pager_Btm_Prev').removeClass('disabled');
    }
    if (currentPage >= totalPages || totalPages <= 1) {
        $('.pagerLastImage').addClass('disabled');
        $('.pagerNextImage').addClass('disabled');
        $('#Pager_Top_Next').addClass('disabled');
        $('#Pager_Btm_Next').addClass('disabled');
        $('#Pager_Btm_Last').addClass('disabled');
    }
    if (totalPages > 1 && currentPage < totalPages) {
        $('.pagerLastImage').removeClass('disabled');
        $('.pagerNextImage').removeClass('disabled');
        $('#Pager_Top_Next').removeClass('disabled');
        $('#Pager_Btm_Next').removeClass('disabled');
        $('#Pager_Btm_Last').removeClass('disabled');
    }
}

function page(btn, pageType) {
    if (!btn.hasClass('disabled')) {
        togglePageButtonDisable(true);
        var pageChanged = false;
        var $currentPageInput = $("#CurrentPageInput");
        if (pageType == "first") {
            $currentPageInput.val(1);
            pageChanged = true;
        }
        else if (pageType == "prev") {
            $currentPageInput.val(parseInt($currentPageInput.val()) - 1);
            pageChanged = true;
        }
        else if (pageType == "next") {
            $currentPageInput.val(parseInt($currentPageInput.val()) + 1);
            pageChanged = true;
        }
        else if (pageType == "last") {
            $currentPageInput.val(parseInt($('#lblTotalpages').text()));
            pageChanged = true;
        }

        if (pageChanged) {
            pageAndSortGrid();
        }
        else {
            togglePageButtonDisable(false);
        }
    }
}

}

function multiSelectDropDownClick(item) {
    var filterGroup = "";
    var selection = "";
    var labelText = "";
    if ($(item).hasClass("selectAll")) {
        $(item).siblings().find("input[type='checkbox']:visible").each(function () {
            $(this).attr('checked', !$(item).find("input[type='checkbox']")[0].checked);
        });
        $(item).find("input[type='checkbox']").attr('checked', !$(item).find("input[type='checkbox']")[0].checked);

    }
    else {
        var $input = $(item).find("input[type='checkbox'],input[type='radio']");
        if ($input.attr("type") == "checkbox") {
            $input.attr('checked', !$(item).find("input[type='checkbox']")[0].checked);
        }
        else if ($input.attr("type") == "radio") {
            if (!$input[0].checked) {
                $input.attr('checked', true);
            }
            $input.trigger('change');
        }
    }
}

function advancedMultiSelectDropDownClick(item) {
    var filterGroup = "";
    var selection = "";
    var labelText = "";
    var titleText = "";
    var $input = $(item);
    var parent = $input.parent().parent();

    if (parent.hasClass("selectAll")) {
        var checked = $input[0].checked;

        parent.siblings().find("input[type='checkbox']:visible").each(function () {
            filterGroup = parent.closest("ul.multiSelect").attr("id").split("_")[1];
            selection = $(this).parent().find("input").val();
            if (!$(this).is(":checked") && checked) {
                labelText = $(this).parent().find("span").html().trim();
                titleText = $(this).parent().attr("title");
                createTagItItem(selection, labelText, filterGroup, titleText);
            }
            else if ($(this).is(":checked") && !checked) {
                removeTagItItem(filterGroup, selection, null);
            }
            $(this).attr('checked', checked);
        });
    }
    else {
        if ($input.attr("type") == "checkbox") {
            filterGroup = parent.closest("ul.multiSelect").attr("id").split("_")[1];
            selection = $input.val();
            if ($input.is(":checked")) {
                labelText = parent.find("span").html().trim();
                titleText = parent.children("label").attr("title");
                createTagItItem(selection, labelText, filterGroup, titleText);
            }
            else {
                removeTagItItem(filterGroup, selection, null);
            }
        }
        else if ($input.attr("type") == "radio") {
            if (!$input[0].checked) {
                $input.attr('checked', true);
            }
            $input.trigger('change');
        }
    }
}

    $("#optionContainer > ul:visible > li:not(.selectAll) > label > span").each(function () {
        var text = $(this).html().trim().toUpperCase();
        if (text.indexOf(" " + keyword) > -1 ||
            text.indexOf(keyword) == 0 ||
            text.indexOf(" (" + keyword) > -1 ||
            text.indexOf("(" + keyword) > -1) {
            $(this).parent().parent().show();
        }
        else {
            $(this).parent().parent().hide();
        }
    });

function removeFilterSelection(item) {
    var $groupList = $(item).closest(".filterGroupList");
    var filterGroup = $groupList.data("filtergroup");
    var value = $(item).data("selection");
    var label = $(item).data("label").toString();
    var isFrom = label.indexOf("From") > -1
    var toolTipContainer = getToolTipContainer(filterGroup, value);

    if (toolTipContainer.is(":visible")) {
        toolTipContainer.hide();
    }

    switch (filterGroup) {
        case "WorkflowStage":
            $("#Options_WorkflowStage").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "Payer":
            $("#Options_Payer").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "Account":
            $("#Options_Account").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "DenialCategory":
            $("#Options_DenialCategory").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "PatientName":
            $("#txtPatientNameModal").val("");
            $(item).remove();
            break;
        case "ClaimNumber":
            $("#txtClaimNumberModal").val("");
            $(item).remove();
            break;
        case "DenialID":
            $("#txtDenialIDModal").val("");
            $(item).remove();
            break;
        case "BillingProvider":
            $("#Options_BillingProvider").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "RenderingProvider":
            $("#Options_RenderingProvider").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "ClaimType":
            if (value == "Professional") {
                $("#WorkcenterArgs_ProfessionalClaimType").attr("checked", false);
            } else if (value == "Institutional") {
                $("#WorkcenterArgs_InstitutionalClaimType").attr("checked", false);
            }
            $(item).remove();
            break;
        case "DeniedAmount":
            if (isFrom) {
                $("#WorkcenterArgs_DeniedAmountMin").val("");
            }
            else {
                $("#WorkcenterArgs_DeniedAmountMax").val("");
            }
            $("#Options_DeniedAmount > li > input.customDeniedAmount").attr("checked", true);
            $("#customDeniedAmount").show();
            $(item).remove();
            break;
        case "DenialDate":
            if (isFrom) {
                $("#WorkcenterArgs_DenialDateFrom").val("");
            }
            else {
                $("#WorkcenterArgs_DenialDateTo").val("");
            }
            $(item).remove();
            break;
        case "ServiceDate":
            if (isFrom) {
                $("#WorkcenterArgs_ServiceDateFrom").val("");
            }
            else {
                $("#WorkcenterArgs_ServiceDateTo").val("");
            }
            $(item).remove();
            break;
        case "ReminderDate":
            $("#modalReminderDate").attr("checked", false);
            $(item).remove();
            break;
        case "FilterPriorityDenials":
            $("#modalFilterPriorityDenials").attr("checked", false);
            $(item).remove();
            break;
        case "LinkedStatus":
            if (value == "Linked") {
                $("#WorkcenterArgs_LinkedDenials").attr("checked", false);
            } else if (value == "Unlinked") {
                $("#WorkcenterArgs_UnlinkedDenials").attr("checked", false);
            }
            $(item).remove();
            break;
        case "CarcCodes":
            $("#Options_CarcCodes").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "RarcCodes":
            $("#Options_RarcCodes").children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + value + "']").attr("checked", false);
            $(item).remove();
            break;
        case "FilterEligibilityResponse":
            $("#modalFilterEligibilityResponse").attr("checked", false);
            $("#modalFilterCoverageDetectionResponse").attr("checked", false);
            $(item).remove();
            break;
        case "FilterCoverageDetectionResponse":
            $("#modalFilterCoverageDetectionResponse").attr("checked", false);
            $("#modalFilterEligibilityResponse").attr("checked", false);
            $(item).remove();
            break;
        default:
            break;
    }
    if ($groupList.find("ul").children().length < 1) {
        $groupList.hide();
    }
}
function createTagItItem(selection, labelText, filterGroup, titleText) {
    var li = document.createElement("li");
    li.setAttribute("class", "tagit-choice hoverToolTip");
    li.setAttribute("data-selection", selection);
    li.setAttribute("data-label", labelText);
    var label = document.createElement("span");
    label.setAttribute("class", "tagit-label");
    label.appendChild(document.createTextNode(labelText));
    var close = document.createElement("span");
    close.setAttribute("class", "tagit-close");
    li.appendChild(label);
    li.appendChild(close);
    if (titleText != null && titleText != "") {
        //li.setAttribute("title", titleText);
        var tooltipContainer = document.createElement("div");
        tooltipContainer.setAttribute("class", "toolTipContainer");
        tooltipContainer.setAttribute("data-selection", selection);
        tooltipContainer.setAttribute("data-filtergroup", filterGroup);
        var tooltipInner = document.createElement("div");
        tooltipInner.setAttribute("class", "toolTipInner shadow");
        tooltipInner.appendChild(document.createTextNode(titleText));
        var tooltipImg = document.createElement("img");
        tooltipImg.setAttribute("class", "arrowImg");
        tooltipImg.setAttribute("src", document.getElementById("ApplicationRoot").href + "Content/images/hint-drop-arrow.png");
        tooltipContainer.appendChild(tooltipInner);
        tooltipContainer.appendChild(tooltipImg);
        $("#advancedSearchModal").append(tooltipContainer);
    }
    var filterGroupList = $("#filterSelection li.filterGroupList[data-filtergroup='" + filterGroup + "']");
    $(filterGroupList).show();
    $(filterGroupList).find("ul").append(li);
    $(li).find(".tagit-close").on("click", function () {
        removeFilterSelection($(this).closest("li"));
    });
    setupAdvancedToolTips(filterGroupList);
}

function removeTagItItem(filterGroup, selection, isFrom) {
    var $filterGroup = $("#filterSelection li.filterGroupList[data-filtergroup='" + filterGroup + "']");
    var $selection = null;
    if (filterGroup == "PatientName" || filterGroup == "ClaimNumber" || filterGroup == "DenialID") {
        $selection = $filterGroup.find("li.tagit-choice");
    }
    else if (filterGroup == "DenialDate" || filterGroup == "ServiceDate" || filterGroup == "DeniedAmount") {
        if (isFrom == true) {
            $selection = $filterGroup.find("li.tagit-choice[data-label^='From']");
        }
        else {
            $selection = $filterGroup.find("li.tagit-choice[data-label^='To']");
        }
    }
    else {
        $selection = $filterGroup.find("li.tagit-choice[data-selection='" + selection + "']");
    }

    $selection.remove();

    if ($filterGroup.find("ul").children().length < 1) {
        $filterGroup.hide();
    }
}

function confirmDeleteSearch(control, userSavedSearchID) {
    var currentSelectedSearchControl = $(control).parent().parent().siblings(".selectedSearch");
    var currentSelectedSearchID = 0;

    if (currentSelectedSearchControl.length > 0)
        currentSelectedSearchID = currentSelectedSearchControl[0].id.replace("SavedSearch", "");

    $('#areYouSureMessage').text('Are you sure you want to delete this search?');

    $('#areYouSureModal').dialog({
        modal: true,
        resizable: false,
        title: "Confirm Delete",
        buttons: {
            "Yes": {
                click: function () {
                    $.ajax({
                        url: DeleteSavedSearchURL,
                        data: JSON.stringify({ searchIDToDelete: userSavedSearchID, selectedSearchID: currentSelectedSearchID }),
                        contentType: 'application/json; charset=utf-8',
                        type: "POST",
                        success: function (data) {
                            toggleVisible('savedSearchList');
                            toggleArrow($('#btnSavedSearchToggle'));

                            var savedSearchDiv = $("#savedSearchDiv");
                            savedSearchDiv.html(data);
                        }
                    });

                    $(this).dialog('close');
                },
                text: "Yes"
            },
            "No": {
                click: function () {
                    $(this).dialog('close')
                },
                text: "No"
            }
        }
    });

    return false;
}

function CarcRarcAutoCompleteSearch(searchBox) {
    var keyword = $(searchBox).val().toUpperCase();
    var filterType = $("#optionContainer > ul:visible").attr("id");
    var filterGroup = filterType.replace("Options_", "");

    if (keyword.length > 0) {
        $.ajax({
            url: filterGroup == "CarcCodes" ? CarcSearchURL : RarcSearchURL,
            contentType: 'application/json; charset=utf-8',
            data: JSON.stringify({
                keyword: keyword
            }),
            type: "POST",
            success: function (data) {
                var options = $("#" + filterType);
                options.html(data);

                // Set checkboxes to match previously selected filters
                var selectedFilters = getFiltersForAdvancedSearch(filterGroup);
                selectedFilters.forEach(function (element) {
                    options.children("li:not(.selectAll)").children("label").children("input[type='checkbox'][value='" + element + "']").attr("checked", true);
                });

                options.find("li label input[type='checkbox']").on("click", function (e) {
                    advancedMultiSelectDropDownClick($(this));
                });
            },
            error: function (jqXHR, textStatus, errorThrown) {
            }
        });
    }
}

function setupAdvancedToolTips(containerElement) {
    $(".hoverToolTip", containerElement)
        .hover(
            function () {
                var filterGroup = $(this).parents("li").data("filtergroup");
                var value = $(this).data("selection");
                var toolTipContainer = getToolTipContainer(filterGroup, value);

                if (!toolTipContainer.is(":visible")) {
                    showAdvancedToolTip($(this), toolTipContainer);
                }
            },
            function () {
                var filterGroup = $(this).parents("li").data("filtergroup");
                var value = $(this).data("selection");
                var toolTipContainer = getToolTipContainer(filterGroup, value);

                if (toolTipContainer.is(":visible")) {
                    toolTipContainer.hide();
                }
            }
        );
}

function showAdvancedToolTip(toolTipButton, toolTipContainer) {
    if (toolTipContainer.length > 0) {
        var height = toolTipContainer.height();
        var width = toolTipContainer.width() / 2;
        var top = $("#searchModalFilters").position().top + toolTipButton.position().top - height + 20;
        var left = $("#searchModalFilters").position().left + toolTipButton.position().left + (toolTipButton.width() / 2) - width;
        toolTipContainer
            .css('top', top)
            .css('left', left)
            .show();
    }
}

function getToolTipContainer(filterGroup, value) {
    return $("#advancedSearchModal").children(".toolTipContainer[data-filtergroup='" + filterGroup + "'][data-selection='" + value + "']");
}

function showFilterOptions(filterCategoryItem) {
    var filterType = $(filterCategoryItem).data("type");
    var filterLoaded = $(filterCategoryItem).data("loaded");
    var options = $("#Options_" + filterType);

    if (filterLoaded == "False") {
        var loadingDiv = $("#Options_Loading");
        loadingDiv.show();
        $.ajax({
            url: WorkcenterURL + "/Load" + filterType + "s",
            contentType: 'application/json; charset=utf-8',
            type: "POST",
            success: function (data) {
                options.html(data);

                options.find("li label input[type='checkbox']").on("click", function (e) {
                    advancedMultiSelectDropDownClick($(this));
                });

                loadingDiv.hide();
                toggleSearchBox(true);
                options.show();
                $(filterCategoryItem).data("loaded", "True");
            },
            error: function (jqXHR, textStatus, errorThrown) {
                loadingDiv.hide();
            }
        });
    }
    else {
        toggleSearchBox(true);
        options.show();
    }
}

function isNullOrWhitespace(input) {

}
function getGridCount(requestArgs) {
    var selectors = {
        upperCntText: '#upResultsCounter',
        upperHiddenCnt: '#hdnTotalCount',
        upperItemCnt: '#topPagerLabel span.lblItemCounts:last-child',
        lowerTotalPages: '#lblTotalpages',
        lowerItemCnt: '#gridPagerLabel span.lblItemCounts:last-child'
    };

    //don't need to find counts on a claimNumber search
    if (isNullOrWhitespace($("#txtClaimNumber").val().trim())) {
        togglePageButtonDisable(true); //blanket disable paging
        prepareCountInfo(selectors);

        requestArgs.success = function (data) {
            setCountInfo(data, selectors);
            togglePageButtonDisable(false); //blanket enable paging
            setPagerStyles(); //wisely figure out which paging to allow
        };

        $.ajax(requestArgs);
    }
}
function setCountInfo(data, selectors) {
    $(selectors.upperCntText).text(data.RecordCount + " Result" + (data.RecordCount > 0 ? "s" : ""));
    $(selectors.upperHiddenCnt).val(data.RecordCount);
    $(selectors.upperItemCnt).text(data.RecordCount);
    $(selectors.lowerItemCnt).text(data.RecordCount);
    $(selectors.lowerTotalPages).text(data.PageCount);
}

function prepareCountInfo(selectors) {

    $(selectors.upperItemCnt).text("?");
    $(selectors.lowerItemCnt).text("?");
    $(selectors.lowerTotalPages).text("?");
}
