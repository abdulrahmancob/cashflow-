/* this is so we can enable the indexOf functionality for Array's in IE 8 - really, Microsoft? */
if (!Array.prototype.indexOf) {
    Array.prototype.indexOf = function (elt /*, from*/) {
        var len = this.length >>> 0;

        var from = Number(arguments[1]) || 0;
        from = (from < 0)
             ? Math.ceil(from)
             : Math.floor(from);
        if (from < 0)
            from += len;

        for (; from < len; from++) {
            if (from in this &&
                this[from] === elt)
                return from;
        }
        return -1;
    };
}

/* Common methods used in validating char input */

function isAllowableControlKey(e, allowDecimalNotation) {

    var allowableControlKeys = new Array(46, 8, 9, 27, 13);

    if (allowDecimalNotation) {
        allowableControlKeys.push(188);
        allowableControlKeys.push(190);
        allowableControlKeys.push(110);
    }

    return $.inArray(e.keyCode, allowableControlKeys) !== -1
        // Allow: Ctrl+A
        || (e.keyCode == 65 && e.ctrlKey === true)
        // Allow: home, end, left, right
        || (e.keyCode >= 35 && e.keyCode <= 39)
        // Allow: Ctrl+V
        || (e.keyCode === 86 && e.ctrlKey === true)
        // Allow: Ctrl+C
        || (e.keyCode === 67 && e.ctrlKey === true);
}

function configureAlphaInput(e) {
    var key = e.keyCode;

    var isValidKey = key >= 65 && key <= 90;

    if (isValidKey || isAllowableControlKey(e)) return;

    e.preventDefault();
}

function configureNumericInput(e) {

    if (isAllowableControlKey(e, false)) return;

    ensureNumericInput(e);
}

function configureAmountInput(e) {

    if (isAllowableControlKey(e, true)) return;

    ensureNumericInput(e);
}

function configureAlphaNumericInput(e) {
    var key = e.keyCode;   

    var isValidKey = (!e.shiftKey && (key >= 48 && key <= 57) || (key >= 65 && key <= 90) || (key >= 96 && key <= 105));

    if (isValidKey || isAllowableControlKey(e)) return;

    e.preventDefault();
}

function configureAlphaNumericSpaceInput(e) {
    var key = e.keyCode;

    var isValidKey = (!e.shiftKey && (key >= 48 && key <= 57) || (key >= 65 && key <= 90) || (key >= 96 && key <= 105) || (key == 32));

    if (isValidKey || isAllowableControlKey(e)) return;

    e.preventDefault();
}

function configureAlphaNumericSpaceHyphenApostropheInput(e) {
    var key = e.keyCode;

    var isValidKey = (!e.shiftKey && (key >= 48 && key <= 57) || (key >= 65 && key <= 90) || (key >= 96 && key <= 105) ||
        (key == 32) || (!e.shiftKey && key == 222) || (!e.shiftKey && key == 189) || (key == 109));

    if (isValidKey || isAllowableControlKey(e)) return;

    e.preventDefault();
}

function ensureNumericInput(e) {
    // Ensure that it is a number and stop the keypress
    if ((e.shiftKey || (e.keyCode < 48 || e.keyCode > 57)) && (e.keyCode < 96 || e.keyCode > 105)) {
        e.preventDefault();
    }
}

function handleAjaxError(jqXHR, textStatus, errorThrown) {

    var statusCode = jqXHR.status;
    var responseText = jqXHR.responseText;

    if (statusCode == 400 && (responseText == "reason=sessiontimeout" || responseText == "Bad Request")) {
        alert("Your session has timed out. Click OK to log in again.");
        location.reload();
    }
    else {
        alert("An error occurred while processing your request.");
    }

}

function setupHoverToolTips(containerElement) {
    $(".hoverToolTip", containerElement)
        .hover(
            function () {
                if (!$(this).children('.toolTipContainer').is(':visible')) {
                    showToolTip($(this));
                }
            },
            function () {
                if ($(this).children('.toolTipContainer').is(':visible')) {
                    hideToolTip($(this));
                }
            }
        );
}

}

function validateDate(dateInput, allowEmpty) {
    var valid = true;
    if (dateInput != undefined) {
        dateInput = $.trim(dateInput.val());
        if (allowEmpty) {
            valid = (dateInput != 'mm/dd/yyyy');
        }
        else {
            valid = (dateInput != 'mm/dd/yyyy' && dateInput != '');
        }
    }
    else valid = false;
    return valid;
}
